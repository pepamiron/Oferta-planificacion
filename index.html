/** * Fechas de cambio por tienda — Oferta NTC * ---------------------------------------- * Calcula, para cada centro, la fecha en que sus datos cambiaron por última vez. * * Cómo funciona: * 1. Lee la hoja de la oferta y construye una "huella" por centro (un resumen * de todas sus filas y columnas vigiladas). * 2. Compara esa huella con la que guardó la última vez en una hoja oculta. * 3. Si ha cambiado, sella la fecha y hora de hoy para ese centro. * Si no ha cambiado, conserva la fecha anterior. * 4. Escribe el resultado en una hoja de salida: Centro | Estado | Cambiado el. * * Funciona con cualquier forma de actualizar la oferta (a mano, pegado masivo, * IMPORTRANGE u otro script), porque no depende de detectar la edición: compara * el contenido. * * NO modifica la hoja de la oferta. Sobre ella solo lee. * * CONVIVENCIA CON OTROS SCRIPTS * Todos los ficheros de un proyecto de Apps Script comparten el mismo espacio * de nombres. Para no pisar nada, todo lo de aquí va con prefijo "ntc" y este * fichero NO define onOpen(): su menú se instala con un disparador propio, que * convive sin problema con el onOpen que ya tengas en otro fichero. * * INSTALACIÓN * 1. En el Sheet: Extensiones › Apps Script. * 2. "+" junto a Archivos › Script. Nómbralo, p. ej., FechasCambio. * 3. Pega este fichero completo y guarda. * 4. Revisa el bloque NTC_FECHAS_CONFIG de abajo. * 5. En el desplegable de funciones elige ntcInstalar y ejecútala. * Acepta los permisos. Eso crea las hojas, hace el primer sellado, * programa la comprobación horaria y añade el menú "Fechas NTC". * 6. Publica la hoja de salida como CSV: * Archivo › Compartir › Publicar en la web › hoja "Fechas cambio" * › formato CSV › Publicar. Guarda esa URL. * * AVISO: el sellado empieza el día que instalas esto. Los cambios anteriores no * se pueden recuperar; en el primer pase todos los centros quedan con la fecha * de instalación y a partir de ahí ya se diferencian entre sí. */ // ============================== CONFIG ============================== var NTC_FECHAS_CONFIG = { // Nombre EXACTO de la pestaña que contiene la oferta (la que exporta el CSV // que lee la herramienta). Respeta mayúsculas y acentos. HOJA_OFERTA: 'Oferta Negocio', // Fila donde están los títulos de columna (normalmente 1). FILA_CABECERA: 1, // Título de la columna que identifica la tienda. COL_CENTRO: 'Centro', // Columnas que se vigilan para decidir si una tienda ha cambiado. // Déjalo vacío ([]) para vigilar todas menos la de centro. // O concreta las que importan, p. ej.: // ['Fecha', 'Pedidos TM', 'Pedidos TT', 'Vehiculos TM', 'Vehiculos TT'] COLS_VIGILADAS: [], // Hoja de salida (se crea sola). Es la que se publica como CSV. HOJA_DESTINO: 'Fechas cambio', // Hoja oculta donde se guardan las huellas. No la toques a mano. HOJA_HUELLAS: '_ntc_huellas', // Zona horaria para las fechas selladas. ZONA: 'Europe/Madrid', // Sellar automáticamente al pegar en la hoja de la oferta. // Salta con cualquier cambio en esa pestaña (pegado, edición, filas nuevas). // Es la opción recomendada si subes la oferta copiando y pegando a mano. AUTO_AL_PEGAR: true, // Comprobación automática cada X horas (1, 2, 4, 6, 8, 12 o 24). // Déjalo en 0 para no programar nada. Con AUTO_AL_PEGAR activo no hace falta. HORAS_ENTRE_COMPROBACIONES: 0, // Salvaguarda contra pegados a medias. // Si de un pase a otro desaparece más de este porcentaje de centros, el script // se planta: no sella nada y lo deja anotado en el registro. Evita que una // comprobación automática pillada en mitad de un copiar/pegar marque media red // como "nueva". Ponlo a 100 para desactivar la salvaguarda. GUARDA_CAIDA_PCT: 20 }; // ========================== INSTALACIÓN ============================ /** Ejecuta esto UNA VEZ a mano: prepara todo y programa los disparadores. */ function ntcInstalar() { var r = ntcActualizarFechasDeCambio(); ntcBorrarDisparadores_(); var ss = SpreadsheetApp.getActiveSpreadsheet(); var horas = NTC_FECHAS_CONFIG.HORAS_ENTRE_COMPROBACIONES; if (horas > 0) { var t = ScriptApp.newTrigger('ntcActualizarFechasDeCambio').timeBased(); if (horas >= 24) t.everyDays(Math.round(horas / 24)).atHour(6); else t.everyHours(horas); t.create(); } if (NTC_FECHAS_CONFIG.AUTO_AL_PEGAR) { ScriptApp.newTrigger('ntcAlCambiarHoja') .forSpreadsheet(ss) .onChange() .create(); } // Menú por disparador instalable, para no necesitar un onOpen() propio. ScriptApp.newTrigger('ntcCrearMenu') .forSpreadsheet(ss) .onOpen() .create(); ntcCrearMenu(); var msg = 'Instalado.\n\n' + 'Centros detectados: ' + r.total + '\n' + 'Sellados en este pase: ' + r.cambiados + '\n\n' + (NTC_FECHAS_CONFIG.AUTO_AL_PEGAR ? 'Se sellará solo cada vez que pegues en "' + NTC_FECHAS_CONFIG.HOJA_OFERTA + '".\n\n' : (horas > 0 ? 'Se comprobará cada ' + horas + ' h.\n\n' : 'Sin automatismo: usa el menú "Fechas NTC › Comprobar cambios ahora".\n\n')) + 'Siguiente paso: publica la hoja "' + NTC_FECHAS_CONFIG.HOJA_DESTINO + '" como CSV.'; try { SpreadsheetApp.getUi().alert(msg); } catch (e) { Logger.log(msg); } } /** Quita los disparadores de este script (si alguna vez quieres pararlo). */ function ntcDesinstalar() { ntcBorrarDisparadores_(); var msg = 'Automatismos eliminados. Las fechas ya selladas se conservan, y ' + 'puedes seguir usando "Comprobar cambios ahora" a mano.'; try { SpreadsheetApp.getUi().alert(msg); } catch (e) { Logger.log(msg); } } /** Borra solo los disparadores de ESTE script, nunca los de otros ficheros. */ function ntcBorrarDisparadores_() { var mios = { ntcActualizarFechasDeCambio: 1, ntcCrearMenu: 1, ntcAlCambiarHoja: 1 }; ScriptApp.getProjectTriggers().forEach(function (t) { if (mios[t.getHandlerFunction()]) ScriptApp.deleteTrigger(t); }); } /** * Reacciona a los cambios en la hoja de la oferta (pegado incluido). * * Va por disparador instalable onChange, no por onEdit simple, por dos razones: * dispone de los 6 minutos completos (onEdit corta a los 30 segundos, poco para * 100.000+ filas) y también se entera de las filas insertadas o borradas. * * Si el pegado se hace en varios pasos, cada paso lanza una pasada; el cerrojo * evita que se solapen y la salvaguarda de GUARDA_CAIDA_PCT evita sellar con la * hoja a medias. */ function ntcAlCambiarHoja(e) { var tipos = { EDIT: 1, INSERT_ROW: 1, REMOVE_ROW: 1, INSERT_GRID: 1, OTHER: 1 }; if (e && e.changeType && !tipos[e.changeType]) return; try { var hoja = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (hoja && hoja.getName() !== NTC_FECHAS_CONFIG.HOJA_OFERTA) return; } catch (err) { /* si no se puede saber la hoja, seguimos */ } var cerrojo = LockService.getDocumentLock(); if (!cerrojo.tryLock(1000)) { Logger.log('Ya hay una comprobación en curso: esta se descarta.'); return; } try { var r = ntcActualizarFechasDeCambio(); Logger.log(r.cancelado ? r.motivo : 'Sellado automático: ' + r.cambiados + ' de ' + r.total + ' centros.'); } finally { cerrojo.releaseLock(); } } /** Menú propio. No se llama onOpen a propósito: así no pisa el tuyo. */ function ntcCrearMenu() { SpreadsheetApp.getUi() .createMenu('Fechas NTC') .addItem('Comprobar cambios ahora', 'ntcComprobarAhora') .addSeparator() .addItem('Instalar / reinstalar', 'ntcInstalar') .addItem('Parar automatismos', 'ntcDesinstalar') .addToUi(); } function ntcComprobarAhora() { var r = ntcActualizarFechasDeCambio(); if (r.cancelado) { SpreadsheetApp.getUi().alert(r.motivo); return; } SpreadsheetApp.getUi().alert( 'Comprobado.\n\nCentros: ' + r.total + '\nCon cambios ahora: ' + r.cambiados ); } // ============================ PROCESO ============================== /** Función principal. La llama el disparador horario. */ function ntcActualizarFechasDeCambio() { var C = NTC_FECHAS_CONFIG; var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = ss.getSheetByName(C.HOJA_OFERTA); if (!hoja) { throw new Error('No existe la pestaña "' + C.HOJA_OFERTA + '". Corrige NTC_FECHAS_CONFIG.HOJA_OFERTA con el nombre exacto.'); } var huellasAhora = ntcCalcularHuellas_(hoja); var previo = ntcLeerHuellasGuardadas_(ss); // { centro: {huella, estado, fecha} } // --- Salvaguarda: hoja vacía o a medio pegar --- var nAhora = Object.keys(huellasAhora).length; var nAntes = Object.keys(previo).length; if (nAntes > 0 && nAhora < nAntes * (1 - C.GUARDA_CAIDA_PCT / 100)) { var aviso = 'Comprobación cancelada por seguridad: había ' + nAntes + ' centros y ahora se leen ' + nAhora + '. Parece un pegado a medias o una ' + 'hoja incompleta, así que no se ha sellado ninguna fecha. Si la bajada es ' + 'real, vuelve a lanzar "Comprobar cambios ahora" con la hoja ya completa.'; Logger.log(aviso); return { total: nAntes, cambiados: 0, cancelado: true, motivo: aviso }; } var sello = Utilities.formatDate(new Date(), C.ZONA, 'yyyy-MM-dd HH:mm'); var salida = {}; var cambiados = 0; Object.keys(huellasAhora).forEach(function (centro) { var huella = huellasAhora[centro]; var antes = previo[centro]; if (!antes) { // Centro que no habíamos visto nunca. salida[centro] = { huella: huella, estado: 'nueva', fecha: sello }; cambiados++; } else if (antes.huella !== huella) { // Mismo centro, contenido distinto: se sella hoy. salida[centro] = { huella: huella, estado: 'modificada', fecha: sello }; cambiados++; } else { // Sin cambios: conserva su fecha y su estado anterior. salida[centro] = { huella: huella, estado: antes.estado || '', fecha: antes.fecha }; } }); ntcGuardarHuellas_(ss, salida); ntcEscribirDestino_(ss, salida); return { total: Object.keys(salida).length, cambiados: cambiados }; } /** * Construye la huella de cada centro a partir de sus filas. * * Pensado para hojas grandes (100.000+ filas): * - lee por tandas, no toda la hoja de golpe; * - usa getValues() en vez de getDisplayValues() (bastante más rápido); * - no guarda las filas en memoria: acumula un total por centro según las va * leyendo. La combinación (suma + xor + número de filas) no depende del orden, * así que reordenar la hoja sin tocar datos no cuenta como modificación. */ function ntcCalcularHuellas_(hoja) { var C = NTC_FECHAS_CONFIG; var TANDA = 20000; // filas por lectura var ultimaFila = hoja.getLastRow(); var ultimaCol = hoja.getLastColumn(); if (ultimaFila <= C.FILA_CABECERA || ultimaCol < 1) return {}; var cabecera = hoja.getRange(C.FILA_CABECERA, 1, 1, ultimaCol) .getValues()[0].map(ntcNormalizar_); var iCentro = cabecera.indexOf(ntcNormalizar_(C.COL_CENTRO)); if (iCentro < 0) { throw new Error('No encuentro la columna "' + C.COL_CENTRO + '" en la fila ' + C.FILA_CABECERA + ' de "' + C.HOJA_OFERTA + '".'); } // Índices de las columnas a vigilar. var vigiladas = []; if (C.COLS_VIGILADAS && C.COLS_VIGILADAS.length) { C.COLS_VIGILADAS.forEach(function (nombre) { var i = cabecera.indexOf(ntcNormalizar_(nombre)); if (i < 0) throw new Error('COLS_VIGILADAS: no existe la columna "' + nombre + '".'); vigiladas.push(i); }); } else { for (var c = 0; c < cabecera.length; c++) { if (c !== iCentro && cabecera[c] !== '') vigiladas.push(c); } } var nVig = vigiladas.length; var acc = {}; // centro -> { suma, xor, filas } var primera = C.FILA_CABECERA + 1; for (var ini = primera; ini <= ultimaFila; ini += TANDA) { var alto = Math.min(TANDA, ultimaFila - ini + 1); var vals = hoja.getRange(ini, 1, alto, ultimaCol).getValues(); for (var f = 0; f < vals.length; f++) { var fila = vals[f]; var centro = ntcTexto_(fila[iCentro]); if (!centro) continue; var trozo = ''; for (var k = 0; k < nVig; k++) { trozo += ntcTexto_(fila[vigiladas[k]]) + '|'; } var h = ntcHash_(trozo); var a = acc[centro]; if (!a) { acc[centro] = { suma: h, xor: h, filas: 1 }; } else { a.suma = (a.suma + h) >>> 0; a.xor = (a.xor ^ h) >>> 0; a.filas++; } } vals = null; // deja que se libere antes de la siguiente tanda } var huellas = {}; Object.keys(acc).forEach(function (centro) { var a = acc[centro]; huellas[centro] = a.suma.toString(16) + '-' + a.xor.toString(16) + '-' + a.filas; }); return huellas; } /** Valor de celda a texto estable y barato (sin formateos costosos). */ function ntcTexto_(v) { if (v == null || v === '') return ''; if (v instanceof Date) return String(v.getTime()); if (typeof v === 'number' || typeof v === 'boolean') return String(v); return String(v).trim(); } function ntcNormalizar_(s) { return String(s == null ? '' : s) .trim().toLowerCase() .normalize('NFD').replace(/[\u0300-\u036f]/g, '') .replace(/[^a-z0-9]/g, ''); } /** Huella corta y estable (FNV-1a). Devuelve el entero, no texto. */ function ntcHash_(str) { var h = 0x811c9dc5; for (var i = 0; i < str.length; i++) { h ^= str.charCodeAt(i); h = (h * 0x01000193) >>> 0; } return h >>> 0; } // ========================= LECTURA / ESCRITURA ===================== function ntcLeerHuellasGuardadas_(ss) { var hoja = ss.getSheetByName(NTC_FECHAS_CONFIG.HOJA_HUELLAS); if (!hoja) return {}; var v = hoja.getDataRange().getDisplayValues(); var out = {}; for (var i = 1; i < v.length; i++) { var centro = String(v[i][0] || '').trim(); if (!centro) continue; out[centro] = { huella: String(v[i][1] || ''), estado: String(v[i][2] || ''), fecha: String(v[i][3] || '') }; } return out; } function ntcGuardarHuellas_(ss, datos) { var hoja = ss.getSheetByName(NTC_FECHAS_CONFIG.HOJA_HUELLAS); if (!hoja) { hoja = ss.insertSheet(NTC_FECHAS_CONFIG.HOJA_HUELLAS); hoja.hideSheet(); } var filas = [['Centro', 'Huella', 'Estado', 'Cambiado el']]; Object.keys(datos).sort(ntcOrdenCentros_).forEach(function (c) { filas.push([c, datos[c].huella, datos[c].estado, datos[c].fecha]); }); hoja.clearContents(); hoja.getRange(1, 1, filas.length, 4).setValues(filas); } /** Hoja de salida: lo que leerá la herramienta vía CSV publicado. */ function ntcEscribirDestino_(ss, datos) { var hoja = ss.getSheetByName(NTC_FECHAS_CONFIG.HOJA_DESTINO); if (!hoja) hoja = ss.insertSheet(NTC_FECHAS_CONFIG.HOJA_DESTINO); var filas = [['Centro', 'Estado', 'Cambiado el']]; Object.keys(datos).sort(ntcOrdenCentros_).forEach(function (c) { filas.push([c, datos[c].estado, datos[c].fecha]); }); hoja.clearContents(); hoja.getRange(1, 1, filas.length, 3).setNumberFormat('@'); // texto: CSV limpio hoja.getRange(1, 1, filas.length, 3).setValues(filas); hoja.getRange(1, 1, 1, 3).setFontWeight('bold'); hoja.setFrozenRows(1); hoja.autoResizeColumns(1, 3); } function ntcOrdenCentros_(a, b) { return String(a).localeCompare(String(b), 'es', { numeric: true }); }
